主题：创建一个 Bakery 数据库，字段 productName 和 productDay; 用 generator code 创建 API、测试并提交。

← 返回 Quiz 2 复习总览

操作流程说明（考试时按这个顺序做）

  1. 建数据库：Atlas 或本地 MongoDB 都可以，关键是拿到 URI。Atlas URI 里数据库名建议写 bakery。
  2. 放 URI：新建 .env，写 MONGODB_URI=你的连接串; 确认 .gitignore 忽略 .env。
  3. 写 generator code：第一行 require('dotenv').config(); 然后 express / mongoose / express.json()。
  4. 写 Schema / Model：字段 productName、productDay，都是 String + required + trim; Model 名叫 Product，集合默认 products。
  5. 定义 REST URL：GET /api/products、POST /api/products、GET /api/products/:id、PUT /api/products/:id、DELETE /api/products/:id。
  6. 连接并启动：先 await mongoose.connect(process.env.MONGODB_URI)，成功后再 app.listen(...)。
  7. 测试：Postman/curl 测 GET/POST/PUT/DELETE; 看状态码 200/201/204/400/404; 再去 mongosh 或 Atlas Browse Collections 确认数据真的写入。
  8. 提交：只提交代码与说明，不提交真实 .env 或密码文件。
安全提醒：URI 含密码，只能放 .env 或部署平台的 Environment Variables，不能写进 server.js 或提交到 GitHub。

Generator code 骨架（你必须看懂每一部分）

require('dotenv').config({ quiet: true });
const express = require('express');
const mongoose = require('mongoose');

const app = express();
app.use(express.json());

const productSchema = new mongoose.Schema({
  productName: { type: String, required: true, trim: true },
  productDay:  { type: String, required: true, trim: true }
}, { timestamps: true });

const Product = mongoose.model('Product', productSchema);

app.get('/api/products', async (req, res) => {
  res.json(await Product.find());
});

app.post('/api/products', async (req, res) => {
  const { productName, productDay } = req.body;
  const doc = await Product.create({ productName, productDay });
  res.status(201).json(doc);
});

app.get('/api/products/:id', async (req, res) => {
  const doc = await Product.findById(req.params.id);
  if (!doc) return res.status(404).json({ message: 'Not found.' });
  res.json(doc);
});

async function start() {
  await mongoose.connect(process.env.MONGODB_URI);
  app.listen(process.env.PORT || 3000);
}
start();

编程实操教程：从创建到测试运行通过（Bakery API）

Step 4 — 写 server.js（完整可跑版）

下面是 Bakery API 的完整单文件版本。它包含 Schema / Model、GET / POST / PUT / DELETE、404 和错误处理。复制到 server.js。

require('dotenv').config({ quiet: true });
const express = require('express');
const mongoose = require('mongoose');
const cors = require('cors');

const app = express();
const port = process.env.PORT || 3000;

app.use(cors());
app.use(express.json());

// ------------------------------------------------ Schema / Model
const productSchema = new mongoose.Schema(
  {
    productName: { type: String, required: true, trim: true },
    productDay:  { type: String, required: true, trim: true }
  },
  { timestamps: true }
);

const Product = mongoose.model('Product', productSchema);

// ------------------------------------------------ Health / welcome
app.get('/', (req, res) => {
  res.json({ message: 'Welcome to the Bakery API' });
});

// ------------------------------------------------ GET all + filter
app.get('/api/products', async (req, res, next) => {
  try {
    const filter = {};
    if (req.query.productDay) {
      filter.productDay = req.query.productDay;
    }
    const products = await Product.find(filter).sort({ productName: 1 });
    res.json(products);
  } catch (err) {
    next(err);
  }
});

// ------------------------------------------------ GET one
app.get('/api/products/:id', async (req, res, next) => {
  try {
    const product = await Product.findById(req.params.id);
    if (!product) {
      return res.status(404).json({ message: 'Product not found.' });
    }
    res.json(product);
  } catch (err) {
    next(err);
  }
});

// ------------------------------------------------ POST create
app.post('/api/products', async (req, res, next) => {
  try {
    if (req.body.productName && req.body.productDay) {
      const product = await Product.create({
        productName: req.body.productName,
        productDay: req.body.productDay
      });
      res.status(201).json(product);
    } else {
      res.status(400).json({ message: 'productName and productDay are required.' });
    }
  } catch (err) {
    next(err);
  }
});

// ------------------------------------------------ PUT update
app.put('/api/products/:id', async (req, res, next) => {
  try {
    const product = await Product.findByIdAndUpdate(
      req.params.id,
      { productName: req.body.productName, productDay: req.body.productDay },
      { returnDocument: 'after', runValidators: true }
    );
    if (!product) {
      return res.status(404).json({ message: 'Product not found.' });
    }
    res.json(product);
  } catch (err) {
    next(err);
  }
});

// ------------------------------------------------ DELETE
app.delete('/api/products/:id', async (req, res, next) => {
  try {
    const product = await Product.findByIdAndDelete(req.params.id);
    if (!product) {
      return res.status(404).json({ message: 'Product not found.' });
    }
    res.status(204).end();
  } catch (err) {
    next(err);
  }
});

// ------------------------------------------------ 404 + error handler
app.use((req, res, next) => {
  const err = new Error('Not Found');
  err.status = 404;
  next(err);
});

app.use((err, req, res, next) => {
  if (err.name === 'CastError') {
    return res.status(400).json({ error: { message: 'That is not a valid product id.' } });
  }
  if (err.name === 'ValidationError') {
    return res.status(400).json({ error: { message: err.message } });
  }
  res.status(err.status || 500);
  res.json({ error: { message: err.message } });
});

// ------------------------------------------------ connect then listen
async function start() {
  try {
    await mongoose.connect(process.env.MONGODB_URI);
    console.log('Connected to MongoDB');
    app.listen(port, () => {
      console.log('Bakery API listening on port ' + port);
    });
  } catch (err) {
    console.error('Could not connect to MongoDB:', err.message);
    process.exit(1);
  }
}

start();
如果老师给的是数字 id 而不是默认 _id，把 Schema 改成 id: { type: Number, required: true, unique: true }，并把 findById 全部换成 findOne({ id: Number(req.params.id) })。

Step 5 — 启动服务器

npm start

成功日志：

Connected to MongoDB
Bakery API listening on port 3000
如果报 querySrv ENOTFOUND ...CLUSTER.mongodb.net，说明 .env 里还是模板占位符，必须换成真实 URI。改完 .env 后要 Ctrl+C 重新 npm start。

Step 6 — Postman / curl 测试通过

#请求Body (JSON)预期
1GET http://localhost:3000/—200 {"message":"Welcome to the Bakery API"}
2POST /api/products{"productName":"Sourdough","productDay":"Monday"}201，返回带 _id 的新产品
3GET /api/products—200，数组里有 Sourdough
4GET /api/products?productDay=Monday—200，只返回 Monday 的产品
5GET /api/products/<ID>—200，返回单个产品
6PUT /api/products/<ID>{"productName":"Baguette","productDay":"Tuesday"}200，返回更新后的产品
7DELETE /api/products/<ID>—204
8POST /api/products 少字段{"productName":"Croissant"}400
9GET /api/products/abc—400 CastError
10GET /nope—404 Not Found

<ID> 从第 2 步 POST 返回的 _id 里复制。

Step 7 — 验证真的写进 MongoDB

只看到 API 返回还不够，最好去 Atlas 或 mongosh 再确认一次：

mongosh "mongodb+srv://用户名:密码@cluster1.6qa39ly.mongodb.net/bakery"

use bakery
db.products.find().pretty()
db.products.countDocuments()
db.products.find({ productDay: "Monday" })

如果这里能看到刚才 POST 的产品，就说明 Express → Mongoose → MongoDB 全链路成功。
