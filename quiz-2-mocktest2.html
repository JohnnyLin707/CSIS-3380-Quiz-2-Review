B. Schema / Model

新手写法 / Beginner style
const bookSchema = new mongoose.Schema({
  title: String,
  author: String
});

const Book = mongoose.model('Book', bookSchema);

最短写法，只定义字段类型。/ Shortest version: only field types.

老师教材写法 / Teacher style
const bookSchema = new mongoose.Schema(
  {
    title: { type: String, required: true, trim: true },
    author: { type: String, required: true, trim: true },
    pages: { type: Number, required: true, min: 1 },
    fiction: { type: Boolean, default: false }
  },
  { timestamps: true }
);

module.exports = mongoose.model('Book', bookSchema);

字段规则完整，错误信息更清楚。/ Full validation and automatic timestamps.

C. GET all / 读取全部

新手写法 / Beginner style

老师教材写法 / Teacher style
router.route('/')

老师用 Router + 链式 route，并排序。/ Teacher uses Router and sorting.

D. POST / 创建

新手写法 / Beginner style
app.post('/api/books', async (req, res) => {
  const book = await Book.create(req.body);
  res.json(book);
});

能跑，但状态码和安全性较弱。/ It runs, but lacks status code and validation.

老师教材写法 / Teacher style
router.route('/')
  .post(async (req, res) => {
    const { title, author, pages, fiction } = req.body;
    const book = await Book.create({ title, author, pages, fiction });
    res.status(201).json(book);
  });

取出指定字段并 return 201。/ Pick known fields and return 201.

E. GET one / 读取单条

新手写法 / Beginner style

找不到时返回 null，状态码仍是 200。/ Returns null with 200 if missing.

router.route('/:id')

找不到明确返回 404。/ Missing document means 404.

F. PUT / DELETE / 更新和删除

新手写法 / Beginner style
app.put('/api/books/:id', async (req, res) => {
  const book = await Book.findByIdAndUpdate(
    req.params.id,
    req.body,
    { new: true }
  );
  res.json(book);
});

app.delete('/api/books/:id', async (req, res) => {
  await Book.findByIdAndDelete(req.params.id);
  res.sendStatus(204);
});
老师教材写法 / Teacher style
router.route('/:id')
  .put(async (req, res) => {
    const book = await Book.findByIdAndUpdate(
      req.params.id,
      { title, author, pages, fiction },
      { returnDocument: 'after', runValidators: true }
    );
    res.json(book);
  })
  .delete(async (req, res) => {
    const book = await Book.findByIdAndDelete(req.params.id);
    if (!book) return res.status(404).json({ message: 'Book not found.' });
    res.status(204).end();
  });

Mongoose 9 推荐 returnDocument: 'after'，旧 new: true 已废弃。/ Use returnDocument in Mongoose 9.

G. 启动顺序 / Startup order

新手写法 / Beginner style
mongoose.connect(process.env.MONGODB_URI);
app.listen(3000);

能跑，但没等连接结果。/ It may run before the DB is ready.

老师教材写法 / Teacher style
async function start() {
  try {
    await mongoose.connect(process.env.MONGODB_URI);
    app.listen(port, () => console.log('API is running'));
  } catch (err) {
    console.error(err.message);
    process.exit(1);
  }
}
start();

先连数据库再监听，失败就退出。/ Connect first, then listen; exit on failure.

D. routes/books.js — 老师教材风格路由 / Textbook-style routes

const express = require('express');
const Book = require('../models/Book');

const router = express.Router();

// GET /api/books
router.get('/', async (req, res, next) => {
  try {
    const filter = {};
    if (req.query.fiction) {
      filter.fiction = req.query.fiction === 'true';
    }
    const books = await Book.find(filter).sort({ title: 1 });
    res.json(books);
  } catch (err) {
    next(err);
  }
});

// POST /api/books
router.post('/', async (req, res, next) => {
  try {
    const { title, author, pages, fiction } = req.body;
    const book = await Book.create({ title, author, pages, fiction });
    res.status(201).json(book);
  } catch (err) {
    next(err);
  }
});

// GET /api/books/:id
router.get('/:id', async (req, res, next) => {
  try {
    const book = await Book.findById(req.params.id);
    if (!book) {
      return res.status(404).json({ message: 'Book not found.' });
    }
    res.json(book);
  } catch (err) {
    next(err);
  }
});

// PUT /api/books/:id
router.put('/:id', async (req, res, next) => {
  try {
    const { title, author, pages, fiction } = req.body;
    const book = await Book.findByIdAndUpdate(
      req.params.id,
      { title, author, pages, fiction },
      { returnDocument: 'after', runValidators: true }
    );
    if (!book) {
      return res.status(404).json({ message: 'Book not found.' });
    }
    res.json(book);
  } catch (err) {
    next(err);
  }
});

// DELETE /api/books/:id
router.delete('/:id', async (req, res, next) => {
  try {
    const book = await Book.findByIdAndDelete(req.params.id);
    if (!book) {
      return res.status(404).json({ message: 'Book not found.' });
    }
    res.status(204).end();
  } catch (err) {
    next(err);
  }
});

module.exports = router;

E. server.js — 主入口 / Main entry

require('dotenv').config({ quiet: true });
const express = require('express');
const mongoose = require('mongoose');
const cors = require('cors');
const bookRoutes = require('./routes/books');

const app = express();
const port = process.env.PORT || 3000;

app.use(cors());
app.use(express.json());

app.get('/', (req, res) => {
  res.json({ message: 'Welcome to the Bookstore API' });
});

app.use('/api/books', bookRoutes);

// 404 catch-all
app.use((req, res, next) => {
  const err = new Error('Not Found');
  err.status = 404;
  next(err);
});

// error handler
app.use((err, req, res, next) => {
  if (err.name === 'CastError') {
    return res.status(400).json({ error: { message: 'That is not a valid book id.' } });
  }
  if (err.name === 'ValidationError') {
    return res.status(400).json({ error: { message: err.message } });
  }
  res.status(err.status || 500);
  res.json({ error: { message: err.message } });
});

async function start() {
  try {
    await mongoose.connect(process.env.MONGODB_URI);
    console.log('Connected to MongoDB');
    app.listen(port, () => {
      console.log('Bookstore API listening on port ' + port);
    });
  } catch (err) {
    console.error('Could not connect to MongoDB:', err.message);
    process.exit(1);
  }
}

start();
